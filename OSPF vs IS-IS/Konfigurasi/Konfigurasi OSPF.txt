Konfigurasi OSPF : Protokol OSPF dikonfigurasi pada semua router.

Konfigurasi R1
en
conf t
hostname R1
interface loopback0
 ip address 1.1.1.1 255.255.255.255
 no shutdown
interface e0/0
 ip address 10.10.10.1 255.255.255.252
 no shutdown
interface e0/1
 ip address 11.11.11.1 255.255.255.252
 no shutdown
interface e0/2
 ip address 12.12.12.1 255.255.255.252
 no shutdown
router ospf 1
 router-id 1.1.1.1
 network 1.1.1.1 0.0.0.0 area 0
 network 10.10.10.0 0.0.0.3 area 0
 network 11.11.11.0 0.0.0.3 area 1
 network 12.12.12.0 0.0.0.3 area 2
interface range e0/0 - 2
 ip ospf authentication message-digest
 ip ospf message-digest-key 1 md5 admin

 no shutdown

 no shutdown
interface e0/1.11

 ip address 192.168.11.1 255.255.255.0
 no shutdown
interface e0/2

 network 192.168.11.0 0.0.0.255 area 1
 network 192.168.12.0 0.0.0.255 area 1
 network 192.168.13.0 0.0.0.255 area 1

no passive-interface e0/1.11
no passive-interface e0/2.12
no passive-interface e0/2.13
